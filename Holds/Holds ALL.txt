SELECT 
  pv.barcode AS "patron barcode", 
  'p' || pv.record_num || 'a' AS "Patron Record #",
  pv.home_library_code AS "Home Library",
  iv.barcode AS "item barcode",
  'i' || iv.record_num || 'a' AS "Item Record #",
  h.placed_gmt AS "Date placed",
  h.expires_gmt AS "Hold Expires",
  h.location_code AS "Branch Location", 
  h.pickup_location_code AS "P/U Location", 
  h.is_ir AS "IR", 
  h.ir_pickup_location_code AS "IR P/U Location",
  h.is_ir_converted_request AS "IR Converted Rqst",
  h.ir_print_name AS "IR Print Name",
  h.ir_delivery_stop_name AS "IR Delivery Stop Name",
  h.status AS "Hold Status",
  iv.item_status_code AS "Item Status"

FROM sierra_view.hold h

JOIN sierra_view.item_record ir
  ON h.record_id = ir.record_id

JOIN sierra_view.item_view iv
  ON iv.id = ir.id

JOIN sierra_view.patron_record pr
  ON h.patron_record_id = pr.record_id

JOIN sierra_view.patron_view pv
  ON pv.id = pr.id;
